In [ ]:
features = pd.DataFrame(index=prices.index)

# Rendements décalés
for lag in [1, 2, 3, 5, 10]:
    features[f"return_lag_{lag}"] = returns.shift(lag)

# Volatilité glissante
for window in [5, 10, 20, 40, 60]:
    features[f"volatility_{window}d"] = returns.rolling(window).std()

# Volatilité downside
negative_returns = returns.clip(upper=0)
for window in [10, 20, 40]:
    features[f"downside_volatility_{window}d"] = negative_returns.pow(2).rolling(window).mean().pow(0.5)

# Rendements cumulées glissants
for window in [5, 10, 20, 60]:
    features[f"momentum_{window}d"] = prices["Close"] / prices["Close"].shift(window) - 1

# Drawdown
for window in [20, 60, 120]:
    rolling_peak = prices["Close"].rolling(window).max()
    features[f"drawdown_{window}d"] = prices["Close"] / rolling_peak - 1

# Skewness et kurtosis glissants
for window in [20, 40, 60]:
    features[f"skewness_{window}d"] = returns.rolling(window).skew()
    features[f"kurtosis_{window}d"] = returns.rolling(window).kurt()

# High-low range par rapport au Close
features["daily_range"] = (prices["High"] - prices["Low"]) / prices["Close"]

# Résime de volatilité : Vol à court_terme / Vol à long-terme
features["volatility_ratio"] = returns.rolling(10).std() / (returns.rolling(10).std() + 1e-8)  # Eviter division par 0

# Horizon pour la prédiction : 5 jours ourvés
HORIZON = 5

# Rendements cumulés sur 5 jours
future_return = prices["Close"].shift(-HORIZON) / prices["Close"] - 1

# 70% train, 30% test
split_position = int(len(prices) * 0.7)
training_period_returns = future_return.iloc[:split_position].dropna()

# KRACH : Rendement sur 5 jours est dans 5% les pires cas sur l'échantillon d'entraînement
CRASH_QUANTILE = 0.05
crash_threshold = training_period_returns.quantile(CRASH_QUANTILE)

# Détection des krachs (variable à expliquer du modèle)
target = (future_return <= crash_threshold).astype(float)
target.iloc[-HORIZON:] = np.nan    # Les finaux observations n'ont pas encore du prix futur

print("DEFINITION DU KRACH")
print("-"*50)
print("Période de prédiction :", HORIZON, "jours ourvrés")
print(f"Seuil du krach sur l'échantillon d'entraînement : {crash_threshold:.2%}")

dataset = features.copy()

def resume_cv(search, nom, top=10):
    df = pd.DataFrame(search.cv_results_)
    cols = [c for c in df.columns if c.startswith("param_")] + ["mean_test_score", "std_test_score", "rank_test_score"]
    out = df[cols].sort_values("rank_test_score").head(top).copy()
    out.columns = [c.replace("param_model__", "") for c in out.columns]
    print(f"\n{nom} : top {top} (PR_AUC moyenne en validation)")
    print(out.to_string(index=False))
    print("Meilleurs paramètres :", search.best_params_)
    print(f"Meilleure PR_AUC CV : {search.best_score_:.4f}")
    return out

5. Evaluation des modèles¶

Un seuil fixe à 0,5 n'a pas été retenu : les crashs étant rares, les probabilités estimées restent faibles et les modèles ne sont pas calibrés de la même manière. Le seuil est donc défini comme le 90e percentile des probabilités out-of-fold, ce qui fixe la proportion de dates signalées (10 %) et permet une comparaison équitable entre modèles.

# Prédictions out-of-fold seulement sur le train (pour choisir le seuil)
def oof_predictions(estimator, X, y, cv):
    probs, ys = [], []
    for tr_idx, va_idx in cv.split(X):
        est = clone(estimator)
        est.fit(X.iloc[tr_idx], y.iloc[tr_idx])
        probs.append(est.predict_proba(X.iloc[va_idx])[:, 1])
        ys.append(y.iloc[va_idx].values)
    return np.concatenate(probs), np.concatenate(ys)

oof_logit, y_logit = oof_predictions(search_logit.best_estimator_, X_train, y_train, tscv)
oof_rf, y_rf = oof_predictions(search_rf.best_estimator_, X_train, y_train, tscv)

PART_ALERTE = 0.10    # On surveille les 10% de dates les plus risquées

def evaluer(nom, y_true, prob, seuil=None):
    y_true = np.asarray(y_true)
    prob = np.asarray(prob)
    res = {
        "Modèle": nom,
        "ROC-AUC": roc_auc_score(y_true, prob),
        "PR-AUC": average_precision_score(y_true, prob),
        "Brier": brier_score_loss(y_true, prob),
    }
    if seuil is not None:
        alerte = prob >= seuil
        n_al = alerte.sum()
        captes = (alerte & (y_true == 1)).sum()
        res.update({
            "Taux alertes": alerte.mean(),
            "Krachs captés": f"{captes}/{y_true.sum()}",
            "Précision": captes / n_al if n_al else np.nan,
            "Rappel": captes / y_true.sum() if y_true.sum() else np.nan,
        })
    return res

Sur le jeu de test (27 crashs, soit un taux de base de 1,8 %), la forêt aléatoire obtient la meilleure performance de classement, avec une ROC-AUC de 0,68 et une PR-AUC de 0,038, soit environ deux fois le niveau du hasard (0,018).

Au seuil fixé sur les prédictions out-of-fold, elle déclenche 9,8 % d'alertes, proche de l'objectif de 10 %, et détecte 6 crashs sur 27 (rappel de 22 %), avec une précision de 4,1 %.

La régression logistique obtient une ROC-AUC plus faible (0,64) et ne détecte aucun crash : son seuil, appris sur le train, est supérieur à la plupart des probabilités observées sur le test, ce qui réduit le taux d'alerte à 0,9 %. Ce résultat illustre la sensibilité des seuils fixés sur le passé à un changement de régime.

Enfin, le score de Brier élevé de la forêt (0,089) confirme que ses probabilités ne sont pas calibrées : elles doivent être utilisées pour classer les dates à risque, et non comme des probabilités absolues de crash. Ces conclusions reposent sur un faible nombre d'événements (27) ; un intervalle de confiance sur le rappel et la précision renforcerait leur robustesse.

La régression logistique produit des probabilités faibles et stables, généralement comprises entre 0,02 et 0,06 ; son seuil (0,0855) n'est dépassé que ponctuellement, ce qui explique l'absence de détection sur le test.

La forêt aléatoire présente une dynamique beaucoup plus marquée : ses probabilités s'élèvent lors des phases de tension, notamment au début de 2022, à la fin de 2022, au printemps 2023, au printemps 2025 et au début de 2026, et dépassent alors le seuil de 0,49. Les alertes se concentrent ainsi sur des épisodes de stress plutôt que sur des dates isolées.

Ce comportement traduit une sensibilité du modèle aux régimes de forte volatilité, mais aussi un nombre élevé de signaux non suivis de krach. Les probabilités de la forêt doivent donc être interprétées comme un indicateur relatif de tension, et non comme des probabilités absolues

Conclusion¶

Contexte : Ce projet personnel cherche à estimer la probabilité qu'une journée de bourse soit suivie, dans les cinq jours ouvrés, d'un rendement extrêmement négatif, c'est-à-dire d'un « krach ». Il porte sur l'indice CAC 40, téléchargé sur environ vingt ans de données quotidiennes.

Méthodologie :

  • Variables explicatives : rendements décalés, volatilités glissantes, volatilité baissière, momentum, drawdown, asymétrie et aplatissement, amplitude journalière.
  • Cible : un krach est défini comme un rendement sur 5 jours situé parmi les 5 % les plus mauvais de l'échantillon d'entraînement.
  • Découpage temporel : 70 % des données pour l'entraînement, 30 % pour le test, avec une validation croisée temporelle et un écart de 5 jours pour éviter les fuites d'information.
  • Modèles : régression logistique et forêt aléatoire, optimisés sur la PR-AUC, critère adapté aux événements rares.
  • Seuil d'alerte : fixé au 90e percentile des probabilités hors échantillon sur le train, pour que les deux modèles signalent la même proportion de dates (10 %).

Résultats : Sur le test (27 krachs, soit 1,8 % des jours), la forêt aléatoire obtient une ROC-AUC de 0,68 et une PR-AUC de 0,038, soit environ deux fois le niveau du hasard. Au seuil choisi, elle détecte 6 krachs sur 27 (rappel de 22 %), avec une précision de 4,1 %. La régression logistique obtient une ROC-AUC de 0,64 et ne détecte aucun krach, son seuil appris sur le train étant trop élevé pour la période de test. Les variables de volatilité, en particulier la volatilité baissière, ressortent comme les principaux signaux dans les deux modèles

Limites :

  • Très peu de krachs : 27 sur le test, donc des métriques instables. Un seul krach de plus ou de moins modifie fortement le rappel et la précision.
  • Précision faible : environ 4 % des alertes de la forêt aléatoire sont suivies d'un krach.
  • Probabilités non calibrées : le score de Brier élevé de la forêt montre qu'il faut les lire comme un classement, pas comme des probabilités réelles.

Suites possibles :

  • Intervalles de confiance (bootstrap) sur le rappel et la précision, pour mesurer l'incertitude.
  • Calibration des probabilités (Platt ou isotonic) pour les rendre interprétables.
